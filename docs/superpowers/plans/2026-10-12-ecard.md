```js
// PRNG con semilla (mulberry32). Toda la aleatoriedad de los módulos puros entra por aquí.
export function createRng(seed = Date.now()) {
  let a = seed >>> 0;
  const next = () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
  return {
    next,
    int: (min, max) => min + Math.floor(next() * (max - min + 1)),
    chance: (p) => next() < p,
    pick: (arr) => arr[Math.floor(next() * arr.length)],
  };
}
```

- [ ] **Step 6: Implementar `js/rules.js`**

```js
// Reglas canónicas de la Eカード (manga). Funciones puras sin estado.
import { CONFIG } from './config.js';

export const CARD = Object.freeze({ EMPEROR: 'emperor', CITIZEN: 'citizen', SLAVE: 'slave' });
export const SIDE = Object.freeze({ EMPEROR: 'emperor', SLAVE: 'slave' });

const BEATS = Object.freeze({
  [CARD.EMPEROR]: CARD.CITIZEN,
  [CARD.CITIZEN]: CARD.SLAVE,
  [CARD.SLAVE]: CARD.EMPEROR,
});

/** 1 si gana `a`, -1 si gana `b`, 0 si empatan (ciudadano contra ciudadano). */
export function compareCards(a, b) {
  if (a === b) return 0;
  if (BEATS[a] === b) return 1;
  if (BEATS[b] === a) return -1;
  return 0;
}

export function isSpecial(card) {
  return card !== CARD.CITIZEN;
}

export function specialOf(side) {
  return side === SIDE.EMPEROR ? CARD.EMPEROR : CARD.SLAVE;
}

export function opposite(side) {
  return side === SIDE.EMPEROR ? SIDE.SLAVE : SIDE.EMPEROR;
}

export function startingHand(side) {
  return [specialOf(side), CARD.CITIZEN, CARD.CITIZEN, CARD.CITIZEN, CARD.CITIZEN];
}

/** Partidas 1–3 y 7–9: Kaiji es Emperador. 4–6 y 10–12: Esclavo. */
export function kaijiSideForGame(game) {
  return Math.floor((game - 1) / 3) % 2 === 0 ? SIDE.EMPEROR : SIDE.SLAVE;
}

/** Turnos impares: pone primero el bando Emperador. Pares: el bando Esclavo. */
export function firstPlacerSide(turn) {
  return turn % 2 === 1 ? SIDE.EMPEROR : SIDE.SLAVE;
}

export function maxBet(needle) {
  return CONFIG.NEEDLE_MAX - needle;
}

export function isValidBet(mm, needle) {
  return Number.isInteger(mm) && mm >= 1 && mm <= maxBet(needle);
}

export function payout(side, mm) {
  return mm * (side === SIDE.EMPEROR ? CONFIG.RATE_EMPEROR : CONFIG.RATE_SLAVE);
}

/** 'ok' | 'eardrum' (≥30 mm, tímpano) | 'canals' (≥45 mm, 三半規管). */
export function needleStatus(needle) {
  if (needle >= CONFIG.NEEDLE_MAX) return 'canals';
  if (needle >= CONFIG.NEEDLE_EARDRUM) return 'eardrum';
  return 'ok';
}
```

```js
// Estado puro de una sesión de 12 partidas. Cada función devuelve un estado nuevo (no muta).
import { CONFIG } from './config.js';
import {
  compareCards, startingHand, kaijiSideForGame, opposite, firstPlacerSide,
  isValidBet, payout, needleStatus,
} from './rules.js';

export const PHASE = Object.freeze({
  BET: 'bet',
  TURN: 'turn',
  GAME_END: 'gameEnd',
  SESSION_END: 'sessionEnd',
});

export const ENDING = Object.freeze({
  CANALS: 'canals',
  EARDRUM: 'eardrum',
  BROKE: 'broke',
  SURVIVOR: 'survivor',
  TRIUMPH: 'triumph',
});

function dealGame(game) {
  const kaijiSide = kaijiSideForGame(game);
  return {
    game,
    turn: 1,
    kaijiSide,
    bet: null,
    kaijiHand: startingHand(kaijiSide),
    tonegawaHand: startingHand(opposite(kaijiSide)),
    turns: [],
  };
}

export function createMatch() {
  return {
    phase: PHASE.BET,
    needle: 0,
    yen: 0,
    ending: null,
    history: [],
    lastTurn: null,
    lastGame: null,
    ...dealGame(1),
  };
}

export function placeBet(state, mm) {
  if (state.phase !== PHASE.BET) throw new Error(`placeBet en fase ${state.phase}`);
  if (!isValidBet(mm, state.needle)) {
    throw new RangeError(`apuesta inválida ${mm} con aguja ${state.needle}`);
  }
  return { ...state, bet: mm, phase: PHASE.TURN };
}

/** 'kaiji' | 'tonegawa': quién pone carta primero (先) en el turno actual. */
export function whoPlacesFirst(state) {
  return firstPlacerSide(state.turn) === state.kaijiSide ? 'kaiji' : 'tonegawa';
}

/** Turno 5: solo quedan las especiales; se resuelve sin elección. */
export function isForcedTurn(state) {
  return state.phase === PHASE.TURN && state.turn === 5;
}

export function forcedCards(state) {
  return { kaijiCard: state.kaijiHand[0], tonegawaCard: state.tonegawaHand[0] };
}

export function computeEnding(needle, yen) {
  const status = needleStatus(needle);
  if (status === 'canals') return ENDING.CANALS;
  if (status === 'eardrum') return ENDING.EARDRUM;
  if (yen <= 0) return ENDING.BROKE;
  if (yen >= CONFIG.TRUE_ENDING_YEN) return ENDING.TRIUMPH;
  return ENDING.SURVIVOR;
}

function removeOne(hand, card) {
  const i = hand.indexOf(card);
  if (i < 0) throw new Error(`la carta ${card} no está en la mano`);
  return [...hand.slice(0, i), ...hand.slice(i + 1)];
}

export function playTurn(state, kaijiCard, tonegawaCard) {
  if (state.phase !== PHASE.TURN) throw new Error(`playTurn en fase ${state.phase}`);
  const kaijiHand = removeOne(state.kaijiHand, kaijiCard);
  const tonegawaHand = removeOne(state.tonegawaHand, tonegawaCard);
  const cmp = compareCards(kaijiCard, tonegawaCard);
  const result = cmp === 0 ? 'draw' : cmp > 0 ? 'kaiji' : 'tonegawa';
  const lastTurn = {
    game: state.game,
    turn: state.turn,
    first: whoPlacesFirst(state),
    kaijiCard,
    tonegawaCard,
    result,
  };
  const turns = [...state.turns, lastTurn];

  if (result === 'draw') {
    return { ...state, kaijiHand, tonegawaHand, turns, lastTurn, turn: state.turn + 1 };
  }

  const yenDelta = result === 'kaiji' ? payout(state.kaijiSide, state.bet) : 0;
  const needleDelta = result === 'tonegawa' ? state.bet : 0;
  const yen = state.yen + yenDelta;
  const needle = Math.min(state.needle + needleDelta, CONFIG.NEEDLE_MAX);
  const lastGame = {
    game: state.game,
    kaijiSide: state.kaijiSide,
    bet: state.bet,
    winner: result,
    yenDelta,
    needleDelta,
    turns,
  };
  const perforated = needleStatus(needle) !== 'ok';
  return {
    ...state,
    kaijiHand,
    tonegawaHand,
    turns,
    lastTurn,
    lastGame,
    history: [...state.history, lastGame],
    yen,
    needle,
    phase: perforated ? PHASE.SESSION_END : PHASE.GAME_END,
    ending: perforated ? computeEnding(needle, yen) : null,
  };
}

export function nextGame(state) {
  if (state.phase !== PHASE.GAME_END) throw new Error(`nextGame en fase ${state.phase}`);
  if (state.game >= CONFIG.GAMES) {
    return { ...state, phase: PHASE.SESSION_END, ending: computeEnding(state.needle, state.yen) };
  }
  return { ...state, ...dealGame(state.game + 1), phase: PHASE.BET, lastTurn: null };
}

/** Cartas distintas que Kaiji puede jugar ahora (para la UI y el tiempo agotado). */
export function legalCards(state) {
  return [...new Set(state.kaijiHand)];
}
```

```js
// Modelo del pulso de Kaiji: lo que el aparato del oído chiva al reloj de Tonegawa.
import { CONFIG } from './config.js';
import { isSpecial } from './rules.js';

export const PULSE_MODE = Object.freeze({
  NATURAL: 'natural',
  FAKE_CALM: 'fakeCalm',
  FAKE_PANIC: 'fakePanic',
});

export const SIGNAL = Object.freeze({ SPECIAL: 'special', CITIZEN: 'citizen' });

export function truthOf(card) {
  return isSpecial(card) ? SIGNAL.SPECIAL : SIGNAL.CITIZEN;
}

function flip(signal) {
  return signal === SIGNAL.SPECIAL ? SIGNAL.CITIZEN : SIGNAL.SPECIAL;
}

/** Señal que recibe Tonegawa cuando Kaiji confirma `card` con el modo de pulso `mode`. */
export function readSignal(card, mode, rng) {
  const truth = truthOf(card);
  const natural = () => (rng.chance(CONFIG.P_NATURAL) ? truth : flip(truth));
  if (mode === PULSE_MODE.FAKE_CALM) return rng.chance(CONFIG.P_FAKE) ? SIGNAL.CITIZEN : natural();
  if (mode === PULSE_MODE.FAKE_PANIC) return rng.chance(CONFIG.P_FAKE) ? SIGNAL.SPECIAL : natural();
  return natural();
}

/** Si el monitor debe mostrar el pico de nervios para esta carta y modo. */
export function displaysSpike(card, mode) {
  if (mode === PULSE_MODE.FAKE_CALM) return false;
  if (mode === PULSE_MODE.FAKE_PANIC) return true;
  return card != null && isSpecial(card);
}

/** Pulsaciones por minuto que se dibujan en el ECG (cosmético). */
export function bpm({ needle = 0, bet = 0, spike = false }, rng) {
  const noise = (rng.next() * 2 - 1) * CONFIG.BPM_NOISE;
  return Math.round(
    CONFIG.BPM_BASE
      + CONFIG.BPM_PER_MM_NEEDLE * needle
      + CONFIG.BPM_PER_MM_BET * bet
      + (spike ? CONFIG.BPM_SPIKE : 0)
      + noise,
  );
}
```

```js
// IA de Tonegawa: mejor respuesta bayesiana con ruido softmax.
// Memoria = hábitos de Kaiji por bando (modelo de supervivencia por turno) + fiabilidad de la señal del pulso.
import { CONFIG } from './config.js';
import { CARD, SIDE, opposite, specialOf, firstPlacerSide } from './rules.js';
import { SIGNAL, truthOf } from './pulse.js';

const emptyHabits = () => ({ n: [0, 0, 0, 0], d: [0, 0, 0, 0] });

export function createMemory() {
  return {
    habits: { [SIDE.EMPEROR]: emptyHabits(), [SIDE.SLAVE]: emptyHabits() },
    trust: { corr: 0, tot: 0 },
  };
}

/** P(Kaiji juega su especial en `turn` | aún la tiene). Turno 5 = 1. */
export function hazard(memory, kaijiSide, turn) {
  if (turn >= 5) return 1;
  const { n, d } = memory.habits[kaijiSide];
  const i = turn - 1;
  const prior = 1 / (6 - turn);
  return (d[i] + CONFIG.HABIT_ALPHA * prior) / (n[i] + CONFIG.HABIT_ALPHA);
}

/** Lo que Tonegawa cree que acierta la señal del pulso (prior 0,85). */
export function signalAccuracy(memory) {
  const { corr, tot } = memory.trust;
  return (corr + CONFIG.TRUST_PRIOR_CORRECT) / (tot + CONFIG.TRUST_PRIOR_TOTAL);
}

export function tonegawaPlacesSecond(kaijiSide, turn) {
  return firstPlacerSide(turn) === kaijiSide;
}

/** P(la carta de Kaiji es especial | señal). Sin señal devuelve el prior `h`. */
export function posterior(h, a, signal) {
  if (signal == null) return h;
  const likeSpecial = signal === SIGNAL.SPECIAL ? a : 1 - a;
  const likeCitizen = signal === SIGNAL.SPECIAL ? 1 - a : a;
  const den = h * likeSpecial + (1 - h) * likeCitizen;
  return den === 0 ? h : (h * likeSpecial) / den;
}

/** Probabilidad de victoria de Tonegawa por acción, dada la creencia `b` y el valor de continuar. */
export function actionValues(tonegawaSide, b, vNext) {
  if (tonegawaSide === SIDE.EMPEROR) {
    return { special: 1 - b, citizen: b + (1 - b) * vNext };
  }
  return { special: b, citizen: (1 - b) * vNext };
}

/** V(turn): probabilidad de victoria de Tonegawa al empezar `turn`, jugando óptimo según su modelo. */
export function winValue(memory, kaijiSide, turn) {
  const tSide = opposite(kaijiSide);
  if (turn >= 5) return tSide === SIDE.EMPEROR ? 0 : 1;
  const vNext = winValue(memory, kaijiSide, turn + 1);
  const h = hazard(memory, kaijiSide, turn);
  const best = (b) => {
    const q = actionValues(tSide, b, vNext);
    return Math.max(q.special, q.citizen);
  };
  if (!tonegawaPlacesSecond(kaijiSide, turn)) return best(h);
  const a = signalAccuracy(memory);
  const pSpecialSignal = h * a + (1 - h) * (1 - a);
  return pSpecialSignal * best(posterior(h, a, SIGNAL.SPECIAL))
    + (1 - pSpecialSignal) * best(posterior(h, a, SIGNAL.CITIZEN));
}

/**
 * Elige la carta de Tonegawa.
 * @param {object} ctx { kaijiSide, turn, signal: 'special'|'citizen'|null, tonegawaHand }
 * @returns {{ card: string, explain: object }}
 */
export function decide(memory, { kaijiSide, turn, signal = null, tonegawaHand }, rng) {
  const tSide = opposite(kaijiSide);
  const special = specialOf(tSide);
  if (!tonegawaHand.includes(CARD.CITIZEN)) return { card: special, explain: { forced: true } };
  if (!tonegawaHand.includes(special)) return { card: CARD.CITIZEN, explain: { forced: true } };
  const h = hazard(memory, kaijiSide, turn);
  const a = signalAccuracy(memory);
  const b = posterior(h, a, signal);
  const vNext = winValue(memory, kaijiSide, turn + 1);
  const q = actionValues(tSide, b, vNext);
  const pSpecial = 1 / (1 + Math.exp((q.citizen - q.special) / CONFIG.SOFTMAX_TAU));
  const card = rng.chance(pSpecial) ? special : CARD.CITIZEN;
  return { card, explain: { h, a, b, vNext, q, pSpecial } };
}

/** Al empezar una partida: los hábitos del bando que va a jugar Kaiji se desvanecen un poco. */
export function startGame(memory, kaijiSide) {
  const { n, d } = memory.habits[kaijiSide];
  const k = CONFIG.HABIT_DECAY;
  return {
    ...memory,
    habits: { ...memory.habits, [kaijiSide]: { n: n.map((x) => x * k), d: d.map((x) => x * k) } },
  };
}

/**
 * Tras revelar un turno, Tonegawa aprende.
 * @param {object} obs { kaijiSide, turn, kaijiCard, signal|null, result: 'draw'|'kaiji'|'tonegawa' }
 */
export function observeTurn(memory, { kaijiSide, turn, kaijiCard, signal = null, result }) {
  let { habits, trust } = memory;
  if (turn <= 4) {
    const i = turn - 1;
    const { n, d } = habits[kaijiSide];
    const playedSpecial = kaijiCard !== CARD.CITIZEN;
    habits = {
      ...habits,
      [kaijiSide]: {
        n: n.map((x, j) => (j === i ? x + 1 : x)),
        d: d.map((x, j) => (j === i && playedSpecial ? x + 1 : x)),
      },
    };
  }
  if (signal != null) {
    const correct = signal === truthOf(kaijiCard);
    const burned = !correct && result === 'kaiji';
    const w = burned ? CONFIG.TRUST_BURN_WEIGHT : 1;
    trust = {
      corr: trust.corr * CONFIG.TRUST_DECAY + (correct ? w : 0),
      tot: trust.tot * CONFIG.TRUST_DECAY + w,
    };
  }
  return { habits, trust };
}
```

```js
// Descubrimiento de la trampa del pulso: pistas, deducción (reloj → aparato) y revelación forzada.
// Cada función devuelve { state, event } donde `event` es una clave de lines.js o null.
import { CONFIG } from './config.js';

export const DISCOVERY_EVENT = Object.freeze({
  GLANCE: 'glance',
  HINT_READ: 'hintRead',
  HINT_WATCH: 'hintWatch',
  JUST_A_WATCH: 'justAWatch',
  WATCH_SUSPECT: 'watchSuspect',
  DEVICE_ONLY: 'deviceOnly',
  DISCOVER: 'discover',
  DISCOVER_FORCED: 'discoverForced',
  ALREADY: 'already',
});

export function createDiscovery() {
  return { glances: 0, suspiciousWins: 0, suspectWatch: false, discovered: false, forced: false };
}

/** Tonegawa mira el reloj (siempre que pone segundo). Se narra las primeras veces. */
export function onGlance(state) {
  const next = { ...state, glances: state.glances + 1 };
  const narrate = !state.discovered && next.glances <= CONFIG.NARRATED_GLANCES;
  return { state: next, event: narrate ? DISCOVERY_EVENT.GLANCE : null };
}

/** Tonegawa ganó una partida poniendo segundo en el turno decisivo. */
export function onSuspiciousWin(state) {
  const next = { ...state, suspiciousWins: state.suspiciousWins + 1 };
  if (state.discovered) return { state: next, event: null };
  if (next.suspiciousWins === CONFIG.HINT_READ_AT) return { state: next, event: DISCOVERY_EVENT.HINT_READ };
  if (next.suspiciousWins === CONFIG.HINT_WATCH_AT) return { state: next, event: DISCOVERY_EVENT.HINT_WATCH };
  return { state: next, event: null };
}

export function clickWatch(state) {
  if (state.discovered) return { state, event: DISCOVERY_EVENT.ALREADY };
  if (state.glances === 0) return { state, event: DISCOVERY_EVENT.JUST_A_WATCH };
  return { state: { ...state, suspectWatch: true }, event: DISCOVERY_EVENT.WATCH_SUSPECT };
}

export function clickDevice(state) {
  if (state.discovered) return { state, event: DISCOVERY_EVENT.ALREADY };
  if (!state.suspectWatch) return { state, event: DISCOVERY_EVENT.DEVICE_ONLY };
  return { state: { ...state, discovered: true }, event: DISCOVERY_EVENT.DISCOVER };
}

/** Al empezar la partida `game`: si aún no se ha descubierto y es la 10 o posterior, Kaiji se da cuenta. */
export function checkForcedReveal(state, game) {
  if (state.discovered || game < CONFIG.FORCED_REVEAL_GAME) return { state, event: null };
  return { state: { ...state, discovered: true, forced: true }, event: DISCOVERY_EVENT.DISCOVER_FORCED };
}
```

```js
// Monte Carlo de balance: Kaiji ingenuo / explotador contra Tonegawa, sesiones completas con memoria.
import { test } from 'node:test';
import assert from 'node:assert/strict';
import { createRng } from '../js/rng.js';
import { CARD, SIDE, compareCards, opposite, specialOf, startingHand, kaijiSideForGame } from '../js/rules.js';
import { PULSE_MODE, readSignal } from '../js/pulse.js';
import { createMemory, decide, startGame, observeTurn, tonegawaPlacesSecond } from '../js/tonegawa.js';

const SESSIONS = 1500;

// Política ingenua: elige al azar el turno (1–5) en el que jugará su especial; pulso natural.
const naive = (kaijiSide, rng) => {
  const specialTurn = rng.int(1, 5);
  return { specialTurn, mode: () => PULSE_MODE.NATURAL };
};

// Explotador (solo como Esclavo): juega el Esclavo en un turno en el que pone primero (2 o 4) fingiendo calma.
const exploiter = (kaijiSide, rng) => {
  if (kaijiSide === SIDE.EMPEROR) return naive(kaijiSide, rng);
  const specialTurn = rng.chance(0.5) ? 2 : 4;
  return { specialTurn, mode: (card) => (card === CARD.CITIZEN ? PULSE_MODE.NATURAL : PULSE_MODE.FAKE_CALM) };
};

function simGame(memory, kaijiSide, policy, rng) {
  let mem = startGame(memory, kaijiSide);
  const plan = policy(kaijiSide, rng);
  let tHand = startingHand(opposite(kaijiSide));
  for (let turn = 1; turn <= 5; turn++) {
    if (turn === 5) return { mem, kaijiWon: kaijiSide === SIDE.SLAVE };
    const kaijiCard = turn === plan.specialTurn ? specialOf(kaijiSide) : CARD.CITIZEN;
    const signal = tonegawaPlacesSecond(kaijiSide, turn) ? readSignal(kaijiCard, plan.mode(kaijiCard), rng) : null;
    const { card } = decide(mem, { kaijiSide, turn, signal, tonegawaHand: tHand }, rng);
    tHand.splice(tHand.indexOf(card), 1);
    const cmp = compareCards(kaijiCard, card);
    const result = cmp === 0 ? 'draw' : cmp > 0 ? 'kaiji' : 'tonegawa';
    mem = observeTurn(mem, { kaijiSide, turn, kaijiCard, signal, result });
    if (result !== 'draw') return { mem, kaijiWon: result === 'kaiji' };
  }
  throw new Error('inalcanzable');
}

function winRates(policy, seed) {
  const rng = createRng(seed);
  const wins = { [SIDE.EMPEROR]: 0, [SIDE.SLAVE]: 0 };
  const games = { [SIDE.EMPEROR]: 0, [SIDE.SLAVE]: 0 };
  for (let s = 0; s < SESSIONS; s++) {
    let mem = createMemory();
    for (let g = 1; g <= 12; g++) {
      const side = kaijiSideForGame(g);
      const r = simGame(mem, side, policy, rng);
      mem = r.mem;
      games[side]++;
      if (r.kaijiWon) wins[side]++;
    }
  }
  return { emperor: wins.emperor / games.emperor, slave: wins.slave / games.slave };
}

test('balance: Kaiji ingenuo', () => {
  const r = winRates(naive, 2026);
  console.log('ingenuo', r);
  assert.ok(r.slave >= 0.05 && r.slave <= 0.25, `esclavo ingenuo ${r.slave}`);
  assert.ok(r.emperor >= 0.35 && r.emperor <= 0.75, `emperador ingenuo ${r.emperor}`);
});

test('balance: explotar el pulso como Esclavo mejora claramente', () => {
  const naiveRate = winRates(naive, 77).slave;
  const exploitRate = winRates(exploiter, 77).slave;
  console.log('esclavo ingenuo', naiveRate, 'explotador', exploitRate);
  assert.ok(exploitRate >= naiveRate + 0.1, `explotador ${exploitRate} vs ingenuo ${naiveRate}`);
});
```

```js
// localStorage con red: en modo privado, sin permisos o en Node devuelve el valor por defecto.
const PREFIX = 'ecard:';

function store() {
  try {
    return globalThis.localStorage ?? null;
  } catch {
    return null;
  }
}

export function load(key, fallback) {
  try {
    const raw = store()?.getItem(PREFIX + key);
    return raw == null ? fallback : JSON.parse(raw);
  } catch {
    return fallback;
  }
}

export function save(key, value) {
  try {
    store()?.setItem(PREFIX + key, JSON.stringify(value));
  } catch {
    /* sin almacenamiento: el juego sigue igual */
  }
}

const ENDING_RANK = { canals: 0, eardrum: 1, broke: 2, survivor: 3, triumph: 4 };

/** Fusiona el resultado de una sesión con el récord guardado. Pura. */
export function mergeRecord(record, { yen, ending }) {
  const prev = record ?? { bestYen: 0, bestEnding: null };
  const better = prev.bestEnding == null || ENDING_RANK[ending] > ENDING_RANK[prev.bestEnding];
  return {
    bestYen: Math.max(prev.bestYen, yen),
    bestEnding: better ? ending : prev.bestEnding,
  };
}
```

```js
// Sintetizador Web Audio: todos los sonidos se generan, no hay ficheros.
// Ninguna función lanza si no hay AudioContext o si está en silencio.

let ctx = null;
let master = null;
let muted = false;
let noise = null;
let zawaNodes = null;

const VOLUME = 0.7;

export function initAudio() {
  if (ctx) {
    if (ctx.state === 'suspended') ctx.resume();
    return;
  }
  const AC = globalThis.AudioContext || globalThis.webkitAudioContext;
  if (!AC) return;
  ctx = new AC();
  master = ctx.createGain();
  master.gain.value = muted ? 0 : VOLUME;
  master.connect(ctx.destination);
}

export function setMuted(value) {
  muted = value;
  if (master) master.gain.setTargetAtTime(muted ? 0 : VOLUME, ctx.currentTime, 0.03);
}

export const isMuted = () => muted;
const ready = () => ctx && !muted;

function noiseBuffer() {
  if (noise) return noise;
  noise = ctx.createBuffer(1, ctx.sampleRate * 2, ctx.sampleRate);
  const data = noise.getChannelData(0);
  let last = 0;
  for (let i = 0; i < data.length; i++) {
    const white = Math.random() * 2 - 1;
    last = (last + 0.04 * white) / 1.04; // ruido marrón suave
    data[i] = white * 0.6 + last * 3;
  }
  return noise;
}

function envGain(t, attack, hold, release, peak) {
  const g = ctx.createGain();
  g.gain.setValueAtTime(0.0001, t);
  g.gain.exponentialRampToValueAtTime(peak, t + attack);
  g.gain.setValueAtTime(peak, t + attack + hold);
  g.gain.exponentialRampToValueAtTime(0.0001, t + attack + hold + release);
  g.connect(master);
  return g;
}

function osc(type, freq, t, dur, dest) {
  const o = ctx.createOscillator();
  o.type = type;
  o.frequency.setValueAtTime(freq, t);
  o.connect(dest);
  o.start(t);
  o.stop(t + dur + 0.05);
  return o;
}

function noiseBurst(t, dur, filterType, freq, q, peak) {
  const src = ctx.createBufferSource();
  src.buffer = noiseBuffer();
  const f = ctx.createBiquadFilter();
  f.type = filterType;
  f.frequency.setValueAtTime(freq, t);
  f.Q.value = q;
  const g = envGain(t, 0.005, dur * 0.3, dur * 0.7, peak);
  src.connect(f).connect(g);
  src.start(t, Math.random());
  src.stop(t + dur + 0.05);
  return { src, f };
}

export const sfx = {
  place() {
    if (!ready()) return;
    noiseBurst(ctx.currentTime, 0.09, 'lowpass', 1400, 0.7, 0.5);
  },
  flip() {
    if (!ready()) return;
    const t = ctx.currentTime;
    const { f } = noiseBurst(t, 0.22, 'bandpass', 500, 1.2, 0.45);
    f.frequency.exponentialRampToValueAtTime(3200, t + 0.2);
  },
  tick() {
    if (!ready()) return;
    const t = ctx.currentTime;
    osc('square', 2400, t, 0.02, envGain(t, 0.001, 0.005, 0.02, 0.08));
  },
  thump() {
    if (!ready()) return;
    const t = ctx.currentTime;
    for (const [delay, freq, peak] of [[0, 62, 0.9], [0.17, 52, 0.6]]) {
      const g = envGain(t + delay, 0.01, 0.03, 0.18, peak);
      const o = osc('sine', freq, t + delay, 0.25, g);
      o.frequency.exponentialRampToValueAtTime(freq * 0.6, t + delay + 0.2);
    }
  },
  sting() {
    if (!ready()) return;
    const t = ctx.currentTime;
    const lp = ctx.createBiquadFilter();
    lp.type = 'lowpass';
    lp.frequency.setValueAtTime(2400, t);
    lp.frequency.exponentialRampToValueAtTime(300, t + 1.4);
    lp.connect(envGain(t, 0.01, 0.15, 1.3, 0.35));
    for (const f of [110, 116.5, 155.6, 220, 233]) osc('sawtooth', f, t, 1.5, lp);
  },
  drill(seconds = 1.2) {
    if (!ready()) return;
    const t = ctx.currentTime;
    const dur = Math.max(0.4, seconds);
    const g = envGain(t, 0.05, dur - 0.2, 0.2, 0.32);
    const lfo = ctx.createOscillator();
    const lfoGain = ctx.createGain();
    lfo.frequency.value = 34;
    lfoGain.gain.value = 0.18;
    lfo.connect(lfoGain).connect(g.gain);
    lfo.start(t);
    lfo.stop(t + dur + 0.1);
    const a = osc('sawtooth', 85, t, dur, g);
    a.frequency.linearRampToValueAtTime(160, t + dur);
    const b = osc('square', 170, t, dur, g);
    b.frequency.linearRampToValueAtTime(320, t + dur);
    noiseBurst(t, dur, 'highpass', 2500, 0.5, 0.12);
  },
  win() {
    if (!ready()) return;
    const t = ctx.currentTime;
    [523.25, 659.25, 783.99, 1046.5].forEach((f, i) => {
      osc('triangle', f, t + i * 0.09, 0.5, envGain(t + i * 0.09, 0.01, 0.1, 0.4, 0.3));
    });
  },
  lose() {
    if (!ready()) return;
    const t = ctx.currentTime;
    const g = envGain(t, 0.02, 0.4, 1.2, 0.4);
    const o = osc('sine', 220, t, 1.6, g);
    o.frequency.exponentialRampToValueAtTime(38, t + 1.5);
  },
  tinnitus(seconds = 5) {
    if (!ready()) return;
    const t = ctx.currentTime;
    osc('sine', 8000, t, seconds, envGain(t, 0.05, seconds * 0.4, seconds * 0.6, 0.12));
  },
};

/** Murmullo ざわ…ざわ… continuo: ruido filtrado en tres bandas que respiran. */
export function startZawa() {
  if (!ready() || zawaNodes) return;
  const t = ctx.currentTime;
  const out = ctx.createGain();
  out.gain.setValueAtTime(0.0001, t);
  out.gain.exponentialRampToValueAtTime(0.22, t + 0.8);
  out.connect(master);
  const src = ctx.createBufferSource();
  src.buffer = noiseBuffer();
  src.loop = true;
  const nodes = [src, out];
  for (const [freq, rate] of [[320, 0.7], [760, 1.1], [1500, 1.6]]) {
    const f = ctx.createBiquadFilter();
    f.type = 'bandpass';
    f.frequency.value = freq;
    f.Q.value = 6;
    const g = ctx.createGain();
    g.gain.value = 0.4;
    const lfo = ctx.createOscillator();
    const depth = ctx.createGain();
    lfo.frequency.value = rate;
    depth.gain.value = 0.35;
    lfo.connect(depth).connect(g.gain);
    lfo.start(t);
    src.connect(f).connect(g).connect(out);
    nodes.push(lfo);
  }
  src.start(t);
  zawaNodes = nodes;
}

export function stopZawa() {
  if (!zawaNodes || !ctx) return;
  const [src, out, ...lfos] = zawaNodes;
  const t = ctx.currentTime;
  out.gain.setTargetAtTime(0.0001, t, 0.25);
  src.stop(t + 1.2);
  lfos.forEach((l) => l.stop(t + 1.2));
  zawaNodes = null;
}
```

```js
export function gunya(strength = 1, ms = 1800) {
  if (prefersReducedMotion()) {
    flash('#000', 600);
    return wait(600);
  }
  const stage = $('stage');
  const map = $('gunya-map');
  const noise = $('gunya-noise');
  stage.classList.add('is-gunya');
  const peak = 40 * strength;
  return new Promise((resolve) => {
    const start = performance.now();
    const frame = (now) => {
      const p = Math.min((now - start) / ms, 1);
      map.setAttribute('scale', (Math.sin(p * Math.PI) * peak).toFixed(1));

      } else {
        map.setAttribute('scale', '0');
        stage.classList.remove('is-gunya');
        resolve();
      }
```

```js
    requestAnimationFrame(frame);
  });
}

/**
 * Monitor ECG con barrido. `onBeat(bpm, spike)` se llama en cada latido.
 * @returns {{start(): void, stop(): void, setBpm(n: number): void, setSpike(b: boolean): void}}
 */
export function createEcg(canvas, { onBeat } = {}) {
  const g = canvas.getContext('2d');
  const W = canvas.width;
  const H = canvas.height;
  const base = H * 0.62;
  let bpmValue = 72;
  let spike = false;
  let running = false;
  let raf = 0;
  let last = 0;
  let sinceBeat = 0;
  let x = 0;
  let prevY = base;

  const wave = (s) => {
    const amp = spike ? 1 : 0.6;
    if (s < 0.08) return 0;
    if (s < 0.16) return 0.1 * Math.sin(((s - 0.08) / 0.08) * Math.PI);
    if (s < 0.19) return 0;
    if (s < 0.21) return -0.12 * ((s - 0.19) / 0.02);
    if (s < 0.24) return -0.12 + (amp + 0.12) * ((s - 0.21) / 0.03);
    if (s < 0.27) return amp - (amp + 0.28) * ((s - 0.24) / 0.03);
    if (s < 0.3) return -0.28 + 0.28 * ((s - 0.27) / 0.03);
    if (s < 0.42) return 0;
    if (s < 0.58) return 0.2 * Math.sin(((s - 0.42) / 0.16) * Math.PI);
    return 0;
  };

  const frame = (now) => {
    if (!running) return;
    const dt = Math.min(0.05, (now - last) / 1000);
    last = now;
    const steps = Math.max(1, Math.round(dt * 140));
    g.strokeStyle = '#5cff8a';
    g.lineWidth = 2;
    g.shadowColor = '#5cff8a';
    g.shadowBlur = 6;
    for (let i = 0; i < steps; i++) {
      sinceBeat += dt / steps;
      if (sinceBeat >= 60 / bpmValue) {
        sinceBeat = 0;
        onBeat?.(bpmValue, spike);
      }
      const y = base - wave(sinceBeat) * H * 0.55 + (Math.random() - 0.5) * 1.2;
      g.clearRect(x + 1, 0, 16, H);
      g.beginPath();
      g.moveTo(x, prevY);
      g.lineTo(x + 1, y);
      g.stroke();
      prevY = y;
      x = x + 1 >= W ? 0 : x + 1;
    }
    raf = requestAnimationFrame(frame);
  };

  return {
    start() {
      if (running) return;
      running = true;
      last = performance.now();
      raf = requestAnimationFrame(frame);
    },
    stop() {
      running = false;
      cancelAnimationFrame(raf);
    },
    setBpm(n) {
      bpmValue = Math.max(40, Math.min(200, n));
    },
    setSpike(b) {
      spike = b;
    },
  };
}
```

```js
/**
 * Mano de Kaiji (boca arriba, en abanico).
 * @param {{onSelect?: (card: string, el: HTMLElement) => void}} opts
 */
export function createHand({ onSelect } = {}) {
  const root = $('hand');
  let cards = [];
  let selected = -1;
  let enabled = false;

  function setEnabled(value) {
    enabled = value;
    root.classList.toggle('is-locked', !value);
    for (const el of root.children) el.disabled = !value;
  }

  function render(list) {
    cards = list.slice();
    selected = -1;
    const mid = (cards.length - 1) / 2;
    root.replaceChildren(...cards.map((c, i) => {
      const el = createCard(c, { tag: 'button' });
      el.style.setProperty('--rot', `${(i - mid) * 7}deg`);
      el.style.setProperty('--drop', `${Math.abs(i - mid) * 7}px`);
      el.setAttribute('aria-label', `${i + 1}: ${CARD_NAMES[c].es} (${CARD_NAMES[c].jp})`);
      el.setAttribute('aria-pressed', 'false');
      el.addEventListener('click', () => select(i));
      return el;
    }));
    setEnabled(enabled);
  }

  function select(i) {
    if (!enabled || i < 0 || i >= cards.length) return;
    selected = i;
    [...root.children].forEach((el, j) => {
      el.classList.toggle('is-selected', j === i);
      el.setAttribute('aria-pressed', String(j === i));
    });
    onSelect?.(cards[i], root.children[i]);
  }

  function take() {
    if (selected < 0) return null;
    const [card] = cards.splice(selected, 1);
    render(cards);
    return card;
  }

  function takeRandom(rng) {
    if (!cards.length) return null;
    selected = rng.int(0, cards.length - 1);
    return take();
  }

  return {
    render,
    select,
    setEnabled,
    take,
    takeRandom,
    selected: () => (selected >= 0 ? cards[selected] : null),
    count: () => cards.length,
  };
}

/**
 * Reloj de turno. Se pausa mientras haya una superposición o el diálogo de reglas abierto.
 * @returns {() => void} función para pararlo
 */
export function startTimer(seconds, { onTick, onTimeout } = {}) {
  const el = $('timer');
  const bar = el.querySelector('.timer__bar');
  const text = el.querySelector('.timer__text');
  let left = seconds;
  let lastWhole = Math.ceil(left);
  let stopped = false;
  const paint = () => {
    bar.style.transform = `scaleX(${left / seconds})`;
    text.textContent = `${Math.ceil(left)}s`;
    el.classList.toggle('is-urgent', left <= 10);
  };
  const stop = () => {
    if (stopped) return;
    stopped = true;
    clearInterval(id);
    el.hidden = true;
  };
  el.hidden = false;
  paint();
  const id = setInterval(() => {
    if (!$('overlay').hidden || $('rules').open) return;
    left = Math.max(0, left - 0.25);
    paint();
    const whole = Math.ceil(left);
    if (whole !== lastWhole) {
      lastWhole = whole;
      onTick?.(whole);
    }
    if (left <= 0) {
      stop();
      onTimeout?.();
    }
  }, 250);
  return stop;
}
```

```js
/** Constructor de DOM mínimo. Texto siempre como nodo de texto (nunca HTML). */
export function h(tag, props = {}, ...children) {
  const el = document.createElement(tag);
  for (const [k, v] of Object.entries(props)) {
    if (v == null || v === false) continue;
    if (k === 'class') el.className = v;
    else if (k === 'text') el.textContent = v;
    else if (k === 'style') el.style.cssText = v;
    else if (k.startsWith('on')) el.addEventListener(k.slice(2), v);
    else el.setAttribute(k, v === true ? '' : String(v));
  }
  for (const c of children.flat()) {
    if (c == null || c === false) continue;
    el.append(c instanceof Node ? c : document.createTextNode(String(c)));
  }
  return el;
}
```

```js
async function resolveTurn(prev, kaijiCard, tonegawaCard, signal) {
  await fx.wait(450);
  sfx.flip();
  fx.speedLines();
  await table.reveal(kaijiCard, tonegawaCard);
  const next = playTurn(prev, kaijiCard, tonegawaCard);
  session.memory = observeTurn(session.memory, {
    kaijiSide: prev.kaijiSide, turn: prev.turn, kaijiCard, signal, result: next.lastTurn.result,
  });
  session.state = next;
  if (next.lastTurn.result === 'draw') {
    table.setVerdict('draw', 'あいこ');
    await fx.wait(1100);
    table.clearSlots();
    return;
  }
  await gameEffects(prev, next);
  reactToTrust();
}

async function betPhase() {
  const s = session.state;
  const forced = checkForcedReveal(session.disc, s.game);
  session.disc = forced.state;
  if (forced.event) await discover(true);
  if (s.game > 1 && (s.game - 1) % 3 === 0) await screens.showSideSwap(s.kaijiSide);
  session.memory = startGame(session.memory, s.kaijiSide);
  hud.renderHud(s);
  hud.setOrder(whoPlacesFirst(s));
  table.clearSlots();
  table.renderOpponent(s.tonegawaHand.length);
  session.hand.render(shuffled(s.kaijiHand));
  session.hand.setEnabled(false);
  startZawa();
  const mm = await screens.askBet({
    needle: s.needle, side: s.kaijiSide, game: s.game, onPreview: (v) => hud.setGhost(s.needle, v),

  stopZawa();
  hud.setGhost(s.needle, 0);
  session.state = placeBet(s, mm);
  hud.renderHud(session.state);
  say(line(betLineKey(mm, s.needle), rng));
}

    table.placeInSlot('kaiji', choice.card);
    sfx.place();
    signal = readSignal(choice.card, choice.mode, rng);
    await tonegawaGlance();
    startZawa();
    fx.zawa({ count: 5 });
    await fx.wait(1000 + rng.int(0, 900));
    stopZawa();
    decision = decide(session.memory, { ...ctx, signal }, rng);
    table.placeInSlot('tonegawa');
    table.renderOpponent(s.tonegawaHand.length - 1);
    sfx.place();
    say(line('tonegawaSecond', rng), 1400);
  }
  hush();
  await resolveTurn(s, choice.card, decision.card, signal);
}

async function runSession() {
  session.state = createMatch();
  session.memory = createMemory();
  session.disc = createDiscovery();
  session.suspicion = 0;
  setDiscoveredUI(false);
  hud.setNeedle(0);
  hud.setGhost(0, 0);
  await hud.setYen(0);
  $('stage').hidden = false;

  while (session.state.phase !== PHASE.SESSION_END) {
    const { phase } = session.state;
    if (phase === PHASE.BET) await betPhase();
    else if (phase === PHASE.TURN) await turnPhase();
    else if (phase === PHASE.GAME_END) {
      await fx.wait(400);
      table.clearSlots();
      session.state = nextGame(session.state);
    }
  }

  const s = session.state;
  const record = mergeRecord(load('record', null), { yen: s.yen, ending: s.ending });
  save('record', record);
  session.ecg.stop();
  stopZawa();
  hush();
  if (s.ending === 'eardrum' || s.ending === 'canals') sfx.tinnitus(6);
  await fx.wait(600);
  return screens.showEnding({ ending: s.ending, yen: s.yen, needle: s.needle, history: s.history, record });
}

function updateSoundButton() {
  $('sound').setAttribute('aria-pressed', String(!isMuted()));
}

function wireEvents() {
  $('sound').addEventListener('click', () => {
    initAudio();
    setMuted(!isMuted());
    save('muted', isMuted());
    updateSoundButton();
  });
  $('help').addEventListener('click', () => screens.showRules());
  $('watch').addEventListener('click', () => handleDiscoveryClick(clickWatch));
  $('device').addEventListener('click', () => handleDiscoveryClick(clickDevice));
  $('pulse').addEventListener('change', () => updatePulse(session.hand.selected(), currentMode()));
  const unlock = () => initAudio();
  document.addEventListener('pointerdown', unlock, { once: true });
  document.addEventListener('keydown', unlock, { once: true });
  document.addEventListener('keydown', (e) => {
    if (!$('overlay').hidden || $('rules').open || $('stage').hidden) return;
    if (/^[1-5]$/.test(e.key)) {
      session.hand.select(Number(e.key) - 1);
    } else if (e.key === 'Enter' && !e.target.closest?.('button, input') && !$('play').disabled) {
      $('play').click();
    }
  });
}

async function boot() {
  setMuted(load('muted', false));
  updateSoundButton();
  table.preloadCards();
  session.hand = table.createHand({
    onSelect: (card, el) => {
      session.onSelect?.();
      updatePulse(card, currentMode());
      if (isSpecial(card)) {
        fx.dokun(el);
        sfx.thump();
      }
    },
  });
  session.ecg = hud.createEcg($('ecg'), {
    onBeat: (_, spike) => {
      const s = session.state;
      const b = bpm({ needle: s?.needle ?? 0, bet: s?.bet ?? 0, spike }, rng);
      session.ecg.setBpm(b);
      $('bpm').textContent = String(b);
      if (spike) sfx.thump();
    },
  });
  wireEvents();

  let next = 'title';
  for (;;) {
    if (next === 'title') {
      $('stage').hidden = true;
      await screens.showTitle({ record: load('record', null) });
      initAudio();
      if (!load('prologueSeen', false)) {
        await screens.showPrologue();
        save('prologueSeen', true);
      }
    }
    next = await runSession();
  }
}

if (new URLSearchParams(location.search).has('debug')) globalThis.__ecard = session;

boot();
```

````markdown
- **Emperador** vence a **Ciudadano** · **Ciudadano** vence a **Esclavo** · **Esclavo** vence a **Emperador**.
````
